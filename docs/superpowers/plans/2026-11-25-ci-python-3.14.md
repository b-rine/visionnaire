It must stay in `requirements.txt` (fastapi, grpcio and others still need it on 3.14).

**Branch:** `ci/python-3.14` off `origin/main` at `1ec1dba`. Open a PR; do not merge.

### Task 2 — Regenerate lockfiles under 3.14

Run from the worktree root. The block runs in its own fail-fast subshell, and the tool venv lives in a temp dir outside the checkout that's removed on exit, so `pip-compile` itself runs on 3.14 and the caller's shell is untouched:

```bash
(
  set -euo pipefail
  T=$(mktemp -d); trap 'rm -rf "$T"' EXIT
  uv venv -q --python 3.14 "$T/tools"
  VIRTUAL_ENV="$T/tools" uv pip install -q pip-tools==7.7.0 pip==26.3
  "$T/tools/bin/python" -c 'import sys; assert sys.version_info[:2] == (3, 14)'
  cd backend
  "$T/tools/bin/pip-compile" --output-file=requirements.txt --strip-extras requirements.in
  "$T/tools/bin/pip-compile" --constraint=requirements.txt --output-file=requirements-dev.txt --strip-extras requirements-dev.in
)
```

Production first, then dev (the dev file is constrained by the production one). Don't pass `--upgrade`. Before committing, run the Verification script in lockfile mode (`bash /path/to/verify.sh lockfiles`, steps 1–2). The full script runs after Task 3, because step 8 checks the docs.

## Verification

Save the script outside the checkout and run it from the worktree root:

- `bash /path/to/verify.sh lockfiles` runs steps 1–2 only (before the Task 2 commit).
- `bash /path/to/verify.sh` runs everything (after Task 3).

It stops at the first failure and prints `ALL CHECKS PASSED` only if every step passes. Step 3 deletes and recreates the worktree's `backend/.venv` (gitignored), and leaves the fresh 3.14 venv in place for local development.

Prerequisites:

- `uv`, `curl`, `git`, and `npx` (Node 22+)
- registry access to PyPI, npm, ghcr.io and Docker Hub; uv's Python download source (unless CPython 3.14 is already installed); and the OSV API (osv.dev), which the scanner calls even with `--no-resolve`
- a Docker daemon that can bind-mount the worktree, with a Buildx builder that supports `linux/arm64` (native or QEMU/binfmt)
- port 6333 free

```bash
#!/usr/bin/env bash
set -euo pipefail
BASE=1ec1dba
ROOT=$(git rev-parse --show-toplevel); cd "$ROOT"
TMP=$(mktemp -d)
QID=""
cleanup() {
  local rc=$?
  if [ -n "$QID" ]; then docker rm -f "$QID" >/dev/null 2>&1 || echo "warning: could not remove container $QID" >&2; fi
  rm -rf "$TMP" || true
  exit "$rc"
}
trap cleanup EXIT
pins() { grep -E '^[A-Za-z0-9][A-Za-z0-9._-]*==' "$1" | sort; }
expect_no_match() {  # usage: expect_no_match <description> <grep/git-grep command...>; exit 1 = pass
  local desc=$1; shift
  set +e; "$@"; local rc=$?; set -e
  [ "$rc" = 1 ] || { echo "FAIL: $desc (rc=$rc)" >&2; exit 1; }
}
# Final pins: the base pins plus the addendum's anyio and typing-extensions bumps.
bump() { sed -e 's/^anyio==4\.14\.2$/anyio==4.16.0/' -e 's/^typing-extensions==4\.15\.0$/typing-extensions==4.16.0/' "$1" | sort; }

echo "== 1. pin rule vs $BASE"
for f in requirements.txt requirements-dev.txt; do
  git show "$BASE:backend/$f" > "$TMP/base.$f"
  pins "$TMP/base.$f" > "$TMP/base.$f.pins"
  pins "backend/$f" > "$TMP/new.$f.pins"
  [ "$(sed -n 2p "backend/$f")" = "# This file is autogenerated by pip-compile with Python 3.14" ]
done
for f in requirements.txt requirements-dev.txt; do
  grep -qx 'anyio==4.14.2' "$TMP/base.$f.pins"
  grep -qx 'typing-extensions==4.15.0' "$TMP/base.$f.pins"
  bump "$TMP/base.$f.pins" > "$TMP/expected.$f.pins"
  cmp "$TMP/expected.$f.pins" "$TMP/new.$f.pins"
done

echo "== 2. regeneration is idempotent (in a temp copy; checkout untouched)"
uv venv -q --python 3.14 "$TMP/tools"
VIRTUAL_ENV="$TMP/tools" uv pip install -q pip-tools==7.7.0 pip==26.3
mkdir "$TMP/regen"
cp backend/requirements.in backend/requirements-dev.in backend/requirements.txt backend/requirements-dev.txt "$TMP/regen/"
( cd "$TMP/regen"
  "$TMP/tools/bin/pip-compile" -q --output-file=requirements.txt --strip-extras requirements.in
  "$TMP/tools/bin/pip-compile" -q --constraint=requirements.txt --output-file=requirements-dev.txt --strip-extras requirements-dev.in )
cmp backend/requirements.txt "$TMP/regen/requirements.txt"
cmp backend/requirements-dev.txt "$TMP/regen/requirements-dev.txt"
if [ "${1:-all}" = lockfiles ]; then echo "LOCKFILE CHECKS PASSED"; exit 0; fi

echo "== 3. fresh 3.14 venv + unit tests"
rm -rf backend/.venv     # gitignored (.gitignore:132)
uv venv -q --seed --python 3.14 backend/.venv
VIRTUAL_ENV="$ROOT/backend/.venv" uv pip install -q -r backend/requirements.txt -r backend/requirements-dev.txt
backend/.venv/bin/python -c 'import sys; assert sys.version_info[:2] == (3, 14), sys.version'
backend/.venv/bin/python -m pip check
( cd backend && .venv/bin/python -m pytest -q -m "not integration" -p no:cacheprovider )

echo "== 4. integration tests (fresh Qdrant)"
if (exec 3<>/dev/tcp/127.0.0.1/6333) 2>/dev/null; then echo "FAIL: port 6333 already in use" >&2; exit 1; fi
QID=$(docker run -d --rm -p 6333:6333 qdrant/qdrant:v1.18.3)
ready=0
for i in $(seq 1 30); do
  if curl -sf --max-time 2 localhost:6333/readyz >/dev/null; then ready=1; break; fi
  sleep 1
done
[ "$ready" = 1 ] || { echo "FAIL: qdrant not ready" >&2; exit 1; }
( cd backend && QDRANT_HOST=localhost QDRANT_PORT=6333 .venv/bin/python -m pytest -q -m integration -p no:cacheprovider )
docker rm -f "$QID" >/dev/null; QID=""

echo "== 5. pyright (Trunk's version) on backend source, 3.14 target"
npx -y pyright@1.1.411 --warnings --pythonversion 3.14 backend/app backend/tests

echo "== 6. osv-scanner (Trunk's version + --no-resolve override), each lockfile"
for f in requirements.txt requirements-dev.txt; do
  docker run --rm -v "$ROOT/backend:/src:ro" ghcr.io/google/osv-scanner:v2.4.0 scan source --lockfile="/src/$f" --no-resolve
done

echo "== 7. published wheels (no uv cache) on x86_64 + aarch64; ARM64 backend image builds"
for f in requirements.txt requirements-dev.txt; do
  pins "backend/$f" > "$TMP/probe.in"
  [ -s "$TMP/probe.in" ]
  for plat in x86_64-manylinux_2_28 aarch64-manylinux_2_28; do
    uv pip compile -q --no-cache "$TMP/probe.in" --python-version 3.14 --python-platform "$plat" \
      --only-binary :all: --no-deps --no-header --no-annotate -o "$TMP/probe.out"
    pins "$TMP/probe.out" > "$TMP/probe.out.pins"
    cmp "$TMP/probe.in" "$TMP/probe.out.pins"
  done
done
docker buildx inspect --bootstrap > "$TMP/buildx.txt"
grep -q 'linux/arm64' "$TMP/buildx.txt"
docker buildx build -q --platform linux/arm64 -f backend/Dockerfile backend --output type=cacheonly

echo "== 8. expected values present, stale values gone"
grep -n 'python-version:' .github/workflows/ci.yml > "$TMP/pyver.txt"
printf '%s\n' '20:          python-version: "3.14"' '47:          python-version: "3.14"' '77:          python-version: "3.14"' > "$TMP/pyver.expected"
cmp "$TMP/pyver.expected" "$TMP/pyver.txt"
grep -Fqx -- '- Python 3.14' README.md
for s in '`actions/checkout@v7`' '`actions/setup-python@v7`' '`actions/setup-node@v7`' '`docker/setup-qemu-action@v4`' '`docker/setup-buildx-action@v4`' \
         'needs: [lint, backend, backend-integration, frontend, frontend-e2e]' 'all five CI jobs must pass' \
         'pip-compile --output-file=requirements.txt --strip-extras requirements.in' \
         'pip-compile --constraint=requirements.txt --output-file=requirements-dev.txt --strip-extras requirements-dev.in'; do
  grep -qF -- "$s" CLAUDE.md || { echo "FAIL: CLAUDE.md missing: $s" >&2; exit 1; }
done
git show "$BASE:CLAUDE.md" > "$TMP/base.CLAUDE.md"
[ "$(wc -l < CLAUDE.md)" = "$(wc -l < "$TMP/base.CLAUDE.md")" ] || { echo "FAIL: CLAUDE.md line count changed (a bullet was wrapped or split)" >&2; exit 1; }
grep -F 'pip-compile --output-file=requirements.txt --strip-extras requirements.in' CLAUDE.md > "$TMP/pipc.txt"
grep -qF 'pip-compile --constraint=requirements.txt --output-file=requirements-dev.txt --strip-extras requirements-dev.in' "$TMP/pipc.txt" || { echo "FAIL: pip-compile commands not on one bullet line" >&2; exit 1; }
grep -F '`triton_client._client()` passes an explicit `ssl_context_factory`' CLAUDE.md > "$TMP/tls.txt"
grep -qF 'Python 3.14' "$TMP/tls.txt" || { echo "FAIL: TLS note lacks Python 3.14" >&2; exit 1; }
expect_no_match "stale python versions" git grep -nE 'Python 3\.1[0-3]|python-version: "3\.1[0-3]"' -- . ':!docs/superpowers'
grep -qF '`pip-compile --output-file=requirements.txt --strip-extras requirements.in`' "$TMP/pipc.txt"
grep -qF '`pip-compile --constraint=requirements.txt --output-file=requirements-dev.txt --strip-extras requirements-dev.in`' "$TMP/pipc.txt"
expect_no_match "obsolete pip-compile recipes in CLAUDE.md" grep -nF -e '`pip-compile requirements.in`' -e '`pip-compile requirements-dev.in --output-file requirements-dev.txt`' CLAUDE.md
expect_no_match "stale action refs in CLAUDE.md" grep -nE 'setup-(python|node)@v6|setup-(qemu|buildx)-action@v3|all four CI jobs' CLAUDE.md

echo "ALL CHECKS PASSED"
```
